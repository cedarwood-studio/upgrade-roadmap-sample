     1	# frozen_string_literal: true
     2	
     3	require "socket"
     4	require "tempfile"
     5	require "timeout"
     6	
     7	module ProcessHelpers
     8	
     9	  # Run one of the scripts in script/ as a child process for the duration of
    10	  # the block. The child is killed if it is still running when the block ends.
    11	  def run_script(script, env = {})
    12	    output = Tempfile.new("postal-process")
    13	    pid = Process.spawn(env.transform_values(&:to_s), "bundle", "exec", "ruby", script,
    14	                        chdir: Rails.root.to_s, out: output.path, err: [:child, :out])
    15	    yield pid, output
    16	  ensure
    17	    kill_process(pid) if pid
    18	    output&.close!
    19	  end
    20	
    21	  def kill_process(pid)
    22	    return unless Process.waitpid(pid, Process::WNOHANG).nil?
    23	
    24	    Process.kill("KILL", pid)
    25	    Process.wait(pid)
    26	  rescue Errno::ECHILD
    27	    # The process has already exited and been waited for.
    28	  end
    29	
    30	  # Send TERM to the process and return its exit status
    31	  def terminate_process(pid, timeout: 30)
    32	    Process.kill("TERM", pid)
    33	    Timeout.timeout(timeout) { Process.wait2(pid).last }
    34	  end
    35	
    36	  def free_port
    37	    server = TCPServer.new("127.0.0.1", 0)
    38	    server.addr[1]
    39	  ensure
    40	    server&.close
    41	  end
    42	
    43	  def wait_until(timeout: 60, interval: 0.2)
    44	    deadline = Process.clock_gettime(Process::CLOCK_MONOTONIC) + timeout
    45	    loop do
    46	      result = yield
    47	      return result if result
    48	      raise "timed out after #{timeout} seconds" if Process.clock_gettime(Process::CLOCK_MONOTONIC) > deadline
    49	
    50	      sleep interval
    51	    end
    52	  end
    53	
    54	  def connect_when_listening(port, **options)
    55	    wait_until(**options) do
    56	      TCPSocket.new("127.0.0.1", port)
    57	    rescue Errno::ECONNREFUSED
    58	      nil
    59	    end
    60	  end
    61	
    62	  def health_check(port, **options)
    63	    socket = connect_when_listening(port, **options)
    64	    socket.write("GET /health HTTP/1.0\r\n\r\n")
    65	    socket.read
    66	  ensure
    67	    socket&.close
    68	  end
    69	
    70	end
